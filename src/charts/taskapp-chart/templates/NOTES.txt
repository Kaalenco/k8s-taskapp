  kubectl get secret {{ .Values.ingress.tls.secretName }} -n {{ .Values.global.namespace }}

  To create a self-signed certificate:
  openssl req -x509 -nodes -days 365 -newkey rsa:2048 \
    -keyout tls.key -out tls.crt \
    -subj "/CN={{ (index .Values.ingress.hosts 0).host }}/O=TaskApp"

  kubectl create secret tls {{ .Values.ingress.tls.secretName }} \
    --cert=tls.crt --key=tls.key \
    -n {{ .Values.global.namespace }}
